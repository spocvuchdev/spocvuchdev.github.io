Ticket bot

Members press a button, get a private channel with the staff, and staff can claim and close the ticket.

  • Python
  • discord.py
  • Buttons
  • Permissions
# Opens a private channel only the member and staff can see
@bot.tree.command(name="ticket", description="Open a support ticket")
async def ticket(interaction: discord.Interaction):
    overwrites = {
        interaction.guild.default_role: discord.PermissionOverwrite(view_channel=False),
        interaction.user: discord.PermissionOverwrite(view_channel=True),
    }
    channel = await interaction.guild.create_text_channel(
        f"ticket-{interaction.user.name}", overwrites=overwrites
    )
    await interaction.response.send_message(f"Ticket: {channel.mention}", ephemeral=True)
